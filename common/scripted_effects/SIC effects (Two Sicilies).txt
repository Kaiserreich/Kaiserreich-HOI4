SIC_italian_republic_joins_confederation = {
	every_owned_state = {
		limit = {
			NOT = {
				state = 39 #Southern Tyrol
				state = 103 #Rijeka
				state = 159 #Lombardy
				state = 160 #Venetia
				state = 488 #Trieste
				state = 683 #Gorizia
				state = 736 #Istria
				state = 759 #Friuli
				state = 783 #Trentino
			}
		}
		transfer_state_to = ROOT
	}

	hidden_effect = {
		every_subject_country = {
			limit = { is_italian_tag = no }
			set_temp_variable = { target = THIS }
			ROOT = { puppet_country_without_changing_government = yes }
		}
		every_subject_country = {
			limit = { is_ai = yes }
			overlord = {
				set_autonomy = { target = PREV autonomy_state = autonomy_free }
			}
			SIC_country_joins_confederation = yes
		}
		159 = { add_core_of = LOM }
		160 = { add_core_of = VNC }
		759 = { add_core_of = VNC }
	}

	if = {
		limit = {
			owns_state = 159
			owns_state = 160
		}
		every_owned_state = {
			limit = {
				OR = {
					state = 103 #Rijeka
					state = 160 #Venetia
					state = 488 #Trieste
					state = 683 #Gorizia
					state = 736 #Istria
					state = 759 #Friuli
				}
			}
			transfer_state_to = VNC
		}
		VNC = {
			transfer_technology = yes
			SIC_country_joins_confederation = yes
		}
		transfer_navy = { target = VNC }
		LOM = {
			annex_country = { target = ITA transfer_troops = yes }
			transfer_technology = yes
			SIC_country_joins_confederation = yes
		}
		if = {
			limit = { ITA = { is_ai = no } }
			LOM = { change_tag_from = ITA }
		}
	}
	else_if = {
		limit = { owns_state = 159 }
		every_owned_state = {
			limit = {
				OR = {
					state = 103 #Rijeka
					state = 488 #Trieste
					state = 683 #Gorizia
					state = 736 #Istria
				}
			}
			transfer_state_to = ROOT
		}
		LOM = {
			annex_country = { target = ITA transfer_troops = yes }
			transfer_technology = yes
			SIC_country_joins_confederation = yes
		}
		if = {
			limit = { ITA = { is_ai = no } }
			LOM = { change_tag_from = ITA }
		}
	}
	else_if = {
		limit = { owns_state = 160 }
		VNC = {
			annex_country = { target = ITA transfer_troops = yes }
			transfer_technology = yes
			SIC_country_joins_confederation = yes
		}
		if = {
			limit = { ITA = { is_ai = no } }
			VNC = { change_tag_from = ITA }
		}
	}
}

			set_cosmetic_tag = SRD_piedmont

	}

	get_current_government_type = yes
	ROOT = {
		set_autonomy = {
			target = PREV
			autonomy_state = federated_state
			end_wars = no
		}
	}
	restore_previous_government_type = yes
}

SIC_unlock_italian_annexations = {
	hidden_effect = {
		every_state = {
			limit = { is_italy_or_potential_claim = yes }
			clear_occupy_state = yes
		}
		if = {
			limit = {
				exists = yes
				NOT = { has_country_flag = open_annexation_event }
			}
			recheck_annexations_immediate = yes
		}
	}
}
