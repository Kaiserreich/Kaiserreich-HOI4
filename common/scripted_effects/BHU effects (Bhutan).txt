increase_bhu_feudal_unity = {
	if = {
		limit = { check_variable = { BHU_feudal_unity_change = 3 } }
		custom_effect_tooltip = { localization_key = BHU_variable_change TEXTICON = BHU_variable_increase_3 TYPE = BHU_feudal_unity }
	}
	else_if = {
		limit = { check_variable = { BHU_feudal_unity_change = 2 } }
		custom_effect_tooltip = { localization_key = BHU_variable_change TEXTICON = BHU_variable_increase_2 TYPE = BHU_feudal_unity }
	}
	else = {
		custom_effect_tooltip = { localization_key = BHU_variable_change TEXTICON = BHU_variable_increase_1 TYPE = BHU_feudal_unity }
	}
	# do not go above 20
	if = {
		limit = { check_variable = { BHU_feudal_unity < 21 } }
		set_temp_variable = { BHU_feudal_unity_new = { value = BHU_feudal_unity add = BHU_feudal_unity_change?1 } }
		BHU_get_feudal_unity_idea = yes
		add_to_variable = {
			BHU_feudal_unity = {
				value = BHU_feudal_unity_change?1
				clamp = { min = 0 max = { value = 20 subtract = BHU_feudal_unity } }
			}
		}
	}
}

decrease_bhu_feudal_unity = {
	if = {
		limit = { check_variable = { BHU_feudal_unity_change = 3 } }

				value = BHU_feudal_unity_change?1

			}
		}
	}
}
