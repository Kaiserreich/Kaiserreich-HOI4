italy_risorgimento_decisions = {
	LOM_unite_italy = {
		name = ITA_unite_italy
		icon = italy_flag
		allowed = { always = no }
		available = {
			italy_non_socialist_can_reunite = yes
		}
		fire_only_once = yes
		cost = 0
		ai_will_do = { factor = 100 }
		priority = 100
		complete_effect = {
			log = "[GetLogRoot][GetLogFrom]: Decision complete LOM_unite_italy"
			log = "KR_Event_Logging;Lombardy UNIFIES ITALY"
			FROM = { italy_break_balance = yes }
			set_country_flag = italy_has_reunited_italy
			if = {
				limit = { is_monarchy = yes }
				if = {
					limit = { has_completed_focus = LOM_crown_king }
					set_cosmetic_tag = LOM_ITA_kingdom
				}
				else = {
					set_cosmetic_tag = LOM_ITA_federation
				}
			}
			else = {
				set_cosmetic_tag = LOM_ITA
			}
			add_political_power = 200
			set_capital = { state = 2 remember_old_capital = no }
			every_subject_country = {
				limit = { is_italian_tag = yes }
				italy_general_transfer_effect = yes
				ROOT = { annex_country = { target = PREV transfer_troops = yes } }
			}
			every_state = {
				limit = { is_italy = yes }
				if = {
					limit = {
						owner = {
							is_italian_tag = yes
							is_subject_of = ROOT
						}
					}
					hidden_effect = { remove_claim_by = ROOT }
					add_core_of = ROOT
				}
				else_if = {
					limit = { NOT = { is_core_of = ROOT } }
					add_claim_by = ROOT
				}
			}
			hidden_effect = {
				if = {
					limit = { has_completed_focus = LOM_rule_of_law }
					news_event = italy_events.201
				}
				country_event = { id = italy_events.60 days = 5 random_days = 5 }
				SWI = { country_event = { id = switzerland_events.211 days = 4 random_days = 8 } } #Campione
				ROM = {
					if = {
						limit = { has_country_flag = ROM_cantacuzino_in_ani }
						country_event = { id = romania_events.53 days = 14 } #Volunteers return
					}
				}
			}
		}
		fixed_random_seed = no
	}
