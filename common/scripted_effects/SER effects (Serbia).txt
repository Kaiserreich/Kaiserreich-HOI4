YUG_update_AI_variables_target = {
	set_variable = { #used as targets for AI spending
		ROOT.YUG_AI_target_@THIS = {
			value = ROOT.YUG_ethnicity_pp_requirement_@THIS
			if = { limit = { value = ROOT.YUG_ethnicity_relations_level_@THIS less_than = 10 } add = 1 }
		}
	}
	set_variable = {
		ROOT.YUG_AI_factor_@THIS = {
			value = 3
			divide = ROOT.YUG_ethnicity_pp_requirement_@THIS #used to make the AI focus on smaller requirements first
			clamp = { min = 1 max = 3 }
		}
	}
}

YUG_check_ethnic_situation = {
	set_temp_variable = {
		ethnic_situation = {
			value = 0
			if = {
				limit = {
					value = { value = ROOT.YUG_pc_expenditure_@THIS greater_than = ROOT.YUG_ethnicity_pp_requirement_@THIS }
					and = { value = ROOT.YUG_ethnicity_relations_level_@THIS less_than = 10 }
				}
				add = 1 # 1 = +1 relations level
			}
			else_if = {
				limit = {
					value = { value = ROOT.YUG_pc_expenditure_@THIS less_than = ROOT.YUG_ethnicity_pp_requirement_@THIS }
					and = { value = ROOT.YUG_ethnicity_relations_level_@THIS greater_than = 1 }
				}
				subtract = 1 # -1 = -1 relations level
			}
		}
	}
}
YUG_update_ethnic_situation = {
	if = {
		limit = { check_variable = { checked_ethnicity = SER } }
		add_to_variable = { YUG_serbia_political_power_factor = { value = 0.05 multiply = ethnic_situation } }
		add_to_variable = { YUG_serbia_war_support_factor = { value = 0.025 multiply = ethnic_situation } }
		add_to_variable = { YUG_serbia_army_org_factor = { value = 0.025 multiply = ethnic_situation } }
	}
	else_if = {
		limit = { check_variable = { checked_ethnicity = CRO } }
		add_to_variable = { YUG_croatia_political_power_factor = { value = 0.05 multiply = ethnic_situation } }
		add_to_variable = { YUG_croatia_stability_factor = { value = 0.025 multiply = ethnic_situation } }
		add_to_variable = { YUG_croatia_conscription_factor = { value = 0.05 multiply = ethnic_situation } }
	}
	else_if = {
		limit = { check_variable = { checked_ethnicity = BOS } }
		add_to_variable = { YUG_bosnia_political_power_factor = { value = 0.05 multiply = ethnic_situation } }
		add_to_variable = { YUG_bosnia_stability_factor = { value = 0.025 multiply = ethnic_situation } }
		add_to_variable = { YUG_bosnia_consumer_goods_factor = { value = -0.075 multiply = ethnic_situation } }
	}
	else_if = {
		limit = { check_variable = { checked_ethnicity = SLV } }
		add_to_variable = { YUG_slovenia_political_power_factor = { value = 0.05 multiply = ethnic_situation } }
		add_to_variable = { YUG_slovenia_stability_factor = { value = 0.025 multiply = ethnic_situation } }
		add_to_variable = { YUG_slovenia_industrial_capacity_factory = { value = 0.05 multiply = ethnic_situation } }
	}
	else_if = {
		limit = { check_variable = { checked_ethnicity = ALB } }
		add_to_variable = { YUG_other_political_power_factor = { value = 0.05 multiply = ethnic_situation } }
		add_to_variable = { YUG_other_stability_factor = { value = 0.025 multiply = ethnic_situation } }
		add_to_variable = { YUG_other_production_speed_buildings_factor = { value = 0.05 multiply = ethnic_situation } }
	}

	if = {
		limit = { check_variable = { ethnic_situation > 0 } }
		custom_effect_tooltip = YUG_improve_relations_tt
	}
	else_if = {
		limit = { check_variable = { ethnic_situation < 0 } }
		custom_effect_tooltip = YUG_reduce_relations_tt
	}
	var:checked_ethnicity = {
		add_to_variable = { PREV.YUG_ethnicity_relations_level_@THIS = ethnic_situation }
		set_variable = {
			PREV.YUG_AI_focus_@THIS = {
				value = 10
				divide = PREV.YUG_ethnicity_relations_level_@THIS
				clamp = { min = 1 max = 10 }
			}
		}
		YUG_update_AI_variables_target = yes
	}
}
