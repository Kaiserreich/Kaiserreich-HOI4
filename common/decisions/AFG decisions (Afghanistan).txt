				#Balance of Power
				effect_tooltip = {
					if = {
						limit = { check_variable = { modifier@AFG_monthly_stability_change > 0 } }
						add_stability = modifier@AFG_monthly_stability_change
					}
					set_temp_variable = { AFG_anger_change = AFG_monthly_anger_change_bop }
					AFG_change_anger = yes
					set_temp_variable = { AFG_reform_points_change = AFG_monthly_reform_points_gain_bop }
					AFG_change_reform_points = yes
				}
				# Ideas
				if = {
					limit = { check_expr = { value = modifier@AFG_monthly_reform_points_gain subtract = AFG_monthly_reform_points_gain_bop not_equals = 0 } }
					set_temp_variable = {
						AFG_reform_points_change = {
							value = modifier@AFG_monthly_reform_points_gain
							subtract = AFG_monthly_reform_points_gain_bop
						}
						tooltip = AFG_reform_points_change_bonus_tooltip
					}
				}
				custom_effect_tooltip = tooltip_white_line
				custom_effect_tooltip = AFG_net_change_anger_reform_points_tooltip
				if = {
					limit = { check_variable = { modifier@AFG_monthly_stability_change > 0 } }
					add_stability = modifier@AFG_monthly_stability_change
				}
				set_temp_variable = { AFG_anger_change = modifier@AFG_monthly_anger_change }
				AFG_change_anger = yes
				set_temp_variable = { AFG_reform_points_change = modifier@AFG_monthly_reform_points_gain }
			}
			else = {
				set_temp_variable = { AFG_reform_points_change = { value = 5 add = modifier@AFG_monthly_reform_points_gain } }
			}
			AFG_change_reform_points = yes
			custom_effect_tooltip = tooltip_white_line
			activate_mission = AFG_monthly_reform_progress_mission
		}
		fixed_random_seed = no
	}
