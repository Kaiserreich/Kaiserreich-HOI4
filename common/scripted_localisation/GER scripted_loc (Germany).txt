defined_text = {
	name = GER_GetCardLines
	text = {
		trigger = {
			if = {
				limit = {
					has_variable = GER_black_monday_card:card_draw_value
					set_temp_variable = { total_available_cards = 0 }
					GER_bm_can_draw_cards = no
				}
				set_temp_variable = { not_enough_cards_can_be_drawn = 1 }
			}
			set_temp_variable = {
				GER_bm_cardgame_line_number = {
					value = GER_black_monday_card:card_lines_for_tt
					if = {
						limit = { value = GER_crisis_id equals = GER_black_monday_card:GER_card_type }
						add = 3 #Relief Card lines if super effective
					}
					else_if = {
						limit = {
							value = { value = not_enough_cards_can_be_drawn equals = 1 } #GER_bm_cardgame_cooperation_business_object, GER_bm_cardgame_rathenauplan, GER_bm_cardgame_tbplan
							or = { value = GER_bm_cardgame_discard_array^num less_than = GER_black_monday_card:card_draw_from_discard } #GER_bm_cardgame_corporate_assistance_object
						}
						add = 2 #Show that cards cannot be drawn lines
					}
				}
			}
		}
		localization_key = "GER_card_front_[?GER_bm_cardgame_line_number]_tt"
	}
}
