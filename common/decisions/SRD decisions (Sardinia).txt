italy_foreign_policy_decisions = {
	SRD_join_entente = {
		name = join_faction
		icon = generic_political_discourse

		allowed = { tag = SRD }

		available = {
			italy_balance_has_broken = yes
			OR = {
				second_weltkrieg_has_begun = yes
				at_peace_or_same_enemies_as_FROM = yes
			}
		}

		visible = {
			is_in_faction = no
			is_subject = no
			NFA = { is_in_faction_with = ENT }
			has_socialist_government = no
			FROM = {
				is_faction_leader = yes
				NOT = { has_war_with = ROOT }
				has_socialist_government = no
			}
		}

		targets = { ENT }

		target_root_trigger = {
			NOT = { has_country_flag = ENT_joined_faction_once }
		}
		target_trigger = {
			has_socialist_government = no
			FROM = {
				is_faction_leader = yes
				NOT = { has_war_with = ROOT }
				has_socialist_government = no
			}
		}

		cost = 0
		fire_only_once = yes

		ai_will_do = {
			base = 100
			modifier = {
				factor = 0
				has_war = no
				FROM = { surrender_progress > 0.3 }
			}
			modifier = {
				factor = 0
				has_country_flag = { flag = ITA_recent_peace days < 30 }
			}
		}

		complete_effect = {
			log = "[GetLogRoot][GetLogFrom]: Decision complete SRD_join_entente"
			italy_break_balance = yes
			FROM = { ask_to_join_faction = yes }
			set_country_flag = ENT_joined_faction_once
		}
		fixed_random_seed = no
	}
}

italy_risorgimento_decisions = {
	SRD_attack_italy = {
		name = attack_country
		desc = SRD_attack_italy_desc
		icon = generic_prepare_civil_war

		allowed = { always = no }

		available = {
			NOT = { country_exists = SRI }
			NOT = { has_non_aggression_pact_with = FROM }
			NOT = { has_guaranteed = FROM }
		}

		visible = {
			can_declare_war_on = FROM
			NOT = { FROM = { has_country_flag = ITA_considering_invite_@FROM } } # ROOT will not attack anyone if it is currently inviting someone
			is_subject = no
		}

		target_trigger = {
			can_declare_war_on = FROM
			is_subject = no
		}

		cost = 0
		fire_only_once = no

		ai_will_do = {
			factor = 1
			modifier = {
				factor = 0
				country_is_in_dangerous_war = yes
			}
			modifier = {
				factor = 0
				FROM = { default_attack_AI_trigger = yes }
			}
			modifier = {
				factor = 0
				FROM = { dont_start_major_faction_wars = yes }
			}
			modifier = {
				factor = 0
				second_weltkrieg_has_begun = no
				FROM = {
					OR = {
						is_in_faction = yes
						AND = {
							is_subject = yes
							overlord = { is_italian_tag = no }
						}
						AND = {
							has_guarantor = yes
							any_collection_element = {
								collection = collection:guarantors
								is_italian_tag = no
							}
						}
					}
				}
			}
			modifier = {
				factor = 0
				FROM = { original_tag = SRD }
				any_neighbor_country = {
					is_italian_tag = yes
					has_capitulated = no
				}
			}
			modifier = {
				factor = 0
				OR = {
					has_country_flag = { flag = ITA_currently_inviting value > 0 }	# ROOT will not attack anyone if it is currently inviting someone
					has_country_flag = ITA_considering_invite_@ROOT					# ROOT will not attack anyone if it is currently being invited
				}
			}
			modifier = {
				factor = 0
				# should be asking them to join peacefully
				has_country_flag = SIC_confed
				FROM = {
					is_subject = no
					has_socialist_government = no
					OR = {
						is_in_faction = no
						is_in_faction_with = ROOT
						is_faction_leader = no
					}
					NOT = { has_country_flag = ITA_refuse_join_@ROOT }
				}
			}
			modifier = {
				factor = 0
				has_cosmetic_tag = SRD_ITA
				owns_state = 838 #Liguria
				if = {
					limit = { ITA = { owns_state = 159 } }
					owns_any_state_of = {
						162 #Tuscany
						1004 #Emilia
					}
				}
				else = {
					any_state_of = {
						count = 2
						target = { 159 160 162 1004 } #Lombardy, Venetia, Tuscany, Emilia
						is_owned_and_controlled_by_ROOT_or_subject = yes
					}
				}
				FROM = {
					is_subject = no
					has_socialist_government = no
					OR = {
						is_in_faction = no
						is_in_faction_with = ROOT
						is_faction_leader = no
					}
					NOT = { has_country_flag = ITA_refuse_join_@ROOT }
				}
			}
		}

		complete_effect = {
			log = "[GetLogRoot][GetLogFrom]: Decision complete SRD_attack_italy"
			add_war_support = 0.10
			if = {
				limit = { is_in_faction_with = FROM }
				leave_faction = yes
			}
			declare_war_on = {
				target = FROM
				type = annex_everything
			}
			italy_add_reunification_wars_threat = yes
		}
		fixed_random_seed = no
	}

	SRD_invite_into_kingdom = {
		icon = italy_flag

		allowed = { original_tag = SRD }

		available = {
			is_subject = no
			if = {
				limit = { ITA = { owns_state = 159 } } #Lombardy
				owns_any_state_of = {
					162 #Tuscany
					1004 #Emilia
				}
			}
			else = {
				any_state_of = {
					count = 2
					target = { 159 160 162 1004 } #Lombardy, Venetia, Tuscany, Emilia
					is_owned_and_controlled_by_ROOT_or_subject = yes
				}
			}
			custom_override_tooltip = {
				tooltip = ITA_busy_inviting
				NOT = { has_country_flag = { flag = ITA_currently_inviting value > 0 } }
			}
			controls_all_owned_states = yes
			strength_ratio = { tag = FROM ratio > 1 }
			ic_ratio = { tag = FROM ratio > 1 }
			FROM = {
				if = {
					limit = { has_country_flag = ITA_ask_after_war_@ROOT }
					has_war = no
				}
				else = {
					controls_all_owned_states = yes
				}
				NOT = { has_wargoal_against = ROOT }
				NOT = { is_justifying_wargoal_against = ROOT }
			}
		}

		visible = {
			is_subject = no
			FROM = {
				is_subject = no
				NOT = {
					has_country_flag = ITA_refuse_join_@ROOT
					has_country_flag = { flag = ITA_considering_invite_@FROM value > 0 }	# ROOT cannot invite FROM if it is doing so already
					has_war_with_italian_tag = yes
				}
			}
		}

		cost = 0
		fire_only_once = no

		targets = { ITA SIC PAP VNC EMI TUS LOM }
		target_root_trigger = {
			owns_any_state_of = {
				158 #Piedmont
				838 #Liguria
			}
			NOT = { country_exists = SRI }
		}
		target_trigger = {
			is_subject = no
			FROM = {
				is_subject = no
				NOT = {
					has_country_flag = ITA_refuse_join_@ROOT
					has_country_flag = { flag = ITA_considering_invite_@FROM value > 0 }	# ROOT cannot invite FROM if it is doing so already
					has_war_with_italian_tag = yes
				}
			}
		}

		ai_will_do = {
			factor = 10000
			modifier = {
				factor = 0.1
				NOT = { is_neighbor_of = FROM }
			}
			modifier = {
				factor = 0.1
				OR = {
					tag = ITA
					tag = SIC
					tag = SRD
					tag = PAP
				}
			}
		}

		complete_effect = {
			log = "[GetLogRoot][GetLogFrom]: Decision complete SRD_invite_into_kingdom"
			set_country_flag = { flag = ITA_currently_inviting days = 30 value = 1 }
			country_event = two_sicilies_events.109
			FROM = {
				set_country_flag = { flag = ITA_considering_invite_@FROM days = 30 value = 1 }
				hidden_effect = {
					country_event = two_sicilies_events.110
				}
			}
		}
		fixed_random_seed = no
	}
}
