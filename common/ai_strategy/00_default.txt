default_concentration_target_weight = {
	enable = { always = yes }
	abort_when_not_enabled = yes

	ai_strategy = { #Use as many divisions for force concentration (general cap of 24 or higher)
		type = force_concentration_factor
		value = 500
	}

	ai_strategy = { #By default, do not use concentration pushes
		type = force_concentration_target_weight
		state_trigger = {
			always = yes
		}
		value = -200
	}

	ai_strategy = { #Only push into states that are relatively surrounded to smooth out the front
		type = force_concentration_target_weight
		state_trigger = {
			any_neighbor_state = {
				count = 2
				controller = {
					controls_state = PREV
					is_ally_with = FROM.FROM
				}
			}
		}
		value = 200
	}

	ai_strategy = { #Or if an enemy state is completely surrounded
		type = force_concentration_target_weight
		state_trigger = {
			all_neighbor_state = {
				controller = {
					controls_state = PREV
					is_ally_with = FROM.FROM
				}
			}
		}
		value = 10000
	}
}

default_naval_invasion_range = {
	enable = {
		has_war = yes
		is_major = yes
	}
	abort_when_not_enabled = yes

	ai_strategy = {
		type = invasion_unit_request
		state_trigger = {
			NOT = {
				OR = {
					any_state = {
						controller = { is_ally_with = FROM.FROM }
						OR = {
							is_coastal = yes
							is_island_state = yes
						}
						distance_to = {
							target = ROOT
							value < 3000
						}
					}
					AND = {
						distance_to = {
							target = 629
							value < 1200
						}
					}
					AND = {
						distance_to = {
							target = 630
							value < 1200
						}
					}
					AND = {
						distance_to = {
							target = 385
							value < 800
						}
					}
					AND = {
						distance_to = {
							target = 642
							value < 800
						}
					}
					AND = {
						distance_to = {
							target = 757
							value < 1000
						}
					}
				}
			}
		}
		value = -2000
	}
}

default_refresh_naval_invasion_orders = {
	enable = {
		has_war = yes
		is_major = yes
		OR = {

		value = -9999
