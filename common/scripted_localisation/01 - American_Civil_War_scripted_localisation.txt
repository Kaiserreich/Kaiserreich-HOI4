defined_text = {
	name = GetACWDivisionCount
	text = {
		trigger = {
			if = {
				limit = {
					OR = {
						has_resistance = no
						AND = {
							controller = { tag = ACC }
							is_core_of = USA
							is_owned_by = USA
						}
					}
				}
				set_temp_variable = {
					unit_count = {
						value = state_population_k
						multiply = 0.001
						clamp = { min = 1 max = 50 }
						multiply = 2
						round = yes
					}
				}
			}
			else = {
				set_temp_variable = {
					unit_count = {
						value = state_population_k
						multiply = 0.001
						clamp = { min = 1 max = 50 }
						round = yes
					}
				}
			}
		}
		localization_key = "[?unit_count|G0]"
	}
}
