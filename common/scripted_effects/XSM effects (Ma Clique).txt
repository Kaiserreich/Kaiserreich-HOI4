XSM_add_5_lin = {
	custom_effect_tooltip = XSM_add_5_lin_tooltip
	add_to_variable = {
		XSM_lin_qinghai_control = {
			value = 5
			clamp = { min = { value = XSM_lin_qinghai_control multiply = -1 } max = { value = 100 subtract = XSM_lin_qinghai_control } }
		}
	}
	add_to_variable = {
		XSM_bufang_qinghai_control = {
			value = -5
			clamp = { min = { value = XSM_bufang_qinghai_control multiply = -1 } max = { value = 100 subtract = XSM_bufang_qinghai_control } }
		}
	}
}

XSM_increase_ma_control = {
	add_to_variable = {
		XSM.yushu_control = {
			value = XSM_yushu_control_change
			clamp = { min = 0 max = { value = 100 subtract = XSM.yushu_control } }
		}
		tooltip = XSM_increase_ma_control_tooltip
	}
	set_variable = { TIB.yushu_control = { value = 100 subtract = XSM.yushu_control } }
}

XSM_change_yushu_tensions = {
	add_to_variable = { XSM.XSM_yushu_tensions = XSM_yushu_tensions_change tooltip = XSM_yushu_tension_increase_tooltip }
	hidden_effect = {
		XSM = {
			if = {
				limit = {
					NOT = { check_variable = { XSM_yushu_tensions < 100 } }
					NOT = { has_war_with = TIB }
				}
				country_event = ma_clique_events.200
			}
		}
	}
}
