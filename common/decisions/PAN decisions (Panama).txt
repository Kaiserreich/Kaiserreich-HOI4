PAN_foreign_category = {
	PAN_trade_deal_decision = {
		icon = hol_attract_foreign_investors

		allowed = { always = no } #activated by PAN_on_global_stage

		available = {
			NOT = { has_war_with = FROM }
		}

		visible = {
			FROM = {
				exists = yes
				is_subject = no
			}
			OR = {
				is_subject = no
				is_subject_of = FROM
			}
		}

		cost = 50
		days_remove = 120
		fire_only_once = yes

		ai_will_do = { factor = 100 }

		remove_effect = {
			log = "[GetLogRoot][GetLogFrom]: Decision remove PAN_trade_deal_decision"
			if = {
				limit = {
					OR = {
						has_idea = PAN_economic_recession_3
						has_idea = PAN_economic_recession_2
						has_idea = PAN_economic_recession_1
					}
				}
				PAN_lower_economic_debuff = yes
			}
			else = {
				if = {
					limit = { has_idea = PAN_profit }
					swap_ideas = {
						add_idea = PAN_profit2
						remove_idea = PAN_profit
					}
				}
				else = {
					add_ideas = PAN_profit
				}
			}
		}
		fixed_random_seed = no
	}
}

PAN_industrial_development = {
	PAN_land_development = {
		icon = GFX_decision_generic_agriculture

		allowed = { always = no } #activated by PAN_survey_new_industrial_zones

		available = {
			num_of_civilian_factories_available_for_projects > 0
		}

		modifier = { civilian_factory_use = 1 }

		cost = 60
		days_remove = 70
		fire_only_once = yes

		ai_will_do = { base = 250 }

		remove_effect = {
			log = "[GetLogRoot]: Decision remove PAN_land_development"
			if = {
				limit = { owns_state = 1126 }
				1126 = { increase_state_category_by_one_level = yes }
			}
			else = { 304 = { increase_state_category_by_one_level = yes } }
			custom_effect_tooltip = tooltip_white_line
			activate_targeted_decision = { target = PAN decision = PAN_continue_industrial_expansion }
			activate_targeted_decision = { target = PAN decision = PAN_fuel_for_our_industries }
		}
		fixed_random_seed = no
	}
	PAN_continue_industrial_expansion = {
		icon = GFX_decision_factories

		allowed = { always = no } #activated by PAN_land_development

		available = {
			OR = {
				controls_state = 1126
				controls_state = 304
			}
			num_of_civilian_factories_available_for_projects > 1
		}

		modifier = { civilian_factory_use = 2 }
		cancel_trigger = {
			NOT = { controls_state = 1126 }
			NOT = { controls_state = 304 }
		}

		cost = 120
		days_remove = 70
		fire_only_once = yes

		ai_will_do = { base = 250 }

		cancel_effect = {
			log = "[GetLogRoot]: Decision cancel PAN_continue_industrial_expansion"
			add_political_power = 120
			hidden_effect = {
				activate_targeted_decision = { target = PAN decision = PAN_continue_industrial_expansion }
			}
		}

		remove_effect = {
			log = "[GetLogRoot]: Decision remove PAN_continue_industrial_expansion"
			random_owned_controlled_state = {
				prioritize = { 1126 } #Western Panama
				limit = { free_building_slots = { building = industrial_complex size > 0 include_locked = yes } }
				add_extra_state_shared_building_slots = 1
				add_building_construction = {
					type = industrial_complex
					level = 1
					instant_build = yes
				}
			}
			random_owned_controlled_state = {
				prioritize = { 304 } #Eastern Panama
				limit = { free_building_slots = { building = industrial_complex size > 0 include_locked = yes } }
				add_extra_state_shared_building_slots = 1
				add_building_construction = {
					type = industrial_complex
					level = 1
					instant_build = yes
				}
			}
		}
		fixed_random_seed = no
	}
