### Reform Points and Anger Effects ###
#Set temp vars first
AFG_change_anger = {
	if = {
		limit = {
			has_variable = AFG_anger_level
			NOT = { check_variable = { AFG_anger_change = 0 } }
		}
		# Add Stability loss when anger goes over 50%
		# 50% - 75% Anger, stability falls by -0.5% per unit of anger
		# 75% - 100% Anger, stability falls by -1% per unit of anger.
		set_temp_variable = { AFG_anger_level_after = { value = AFG_anger_level add = AFG_anger_change } }
		if = {
			limit = {
				check_variable = { AFG_anger_level_after > 0.5 }
				check_variable = { AFG_anger_change > 0 }
			}
			set_temp_variable = {
				AFG_relevant_anger_level_change = {
					value = 0
					if = { #Exclude Anger changes that are not relevant to stability loss
						limit = { value = AFG_anger_level less_than = 0.5 }
						add = AFG_anger_level_after
						subtract = 0.5
					}
					else = { add = AFG_anger_change }
				}
			}
			set_temp_variable = {
				AFG_anger_stability_loss = {
					value = 0
					if = {
						limit = { value = AFG_anger_level_after less_than_or_equals = 0.75 }
						# Low stability loss
						add = AFG_relevant_anger_level_change
						multiply = -0.5
					}
					else_if = {
						limit = { value = AFG_anger_level less_than_or_equals = 0.75 }
						# High stability loss
						add = {
							value = AFG_anger_level_after
							subtract = 0.75
						}
						# Low stability loss
						add = {
							value = 0.75
							subtract = AFG_anger_level
							multiply = 0.5
						}
						multiply = -1
					}
					else = {
						# High stability loss
						add = AFG_relevant_anger_level_change
						multiply = -1
					}
				}
			}
			custom_effect_tooltip = AFG_anger_stability_loss_tooltip
			add_stability = AFG_anger_stability_loss
		}

		# Add Anger
		effect_tooltip = { add_to_variable = { AFG_anger_level = AFG_anger_change tooltip = AFG_anger_change_tooltip } }
		hidden_effect = {
			add_to_variable = {
				AFG_anger_level = {
					value = AFG_anger_change
					clamp = { min = { value = AFG_anger_level multiply = -1 } max = { value = 1 subtract = AFG_anger_level } }
				}
			}
		}
	}
}

		}
	}
}

# This is the number of times the effect must be ran to remove the dynmod. Is negative to ensure modifier changes always positive.
# It also influences by how much it changes the modifiers each time, dictated by what the max debuffs were after BM.
@AFG_reform_progress_amount = -30
# Below are constants of the max debuffs of it post black monday
@AFG_national_integration_mobilization_speed_cap = -0.45
@AFG_national_integration_army_org_factor_cap = -0.15
@AFG_national_integration_production_speed_buildings_factor_cap = -0.6
@AFG_national_integration_research_speed_factor_cap = -0.12

AFG_decrease_national_integration = {
	custom_effect_tooltip = {
		localization_key = tooltip_modify_dynmod
		DYNMOD = AFG_national_integration_modifier
	}
	# Mobilisation Speed
	add_to_variable = {
		AFG_national_integration_mobilization_speed = {
			value = @AFG_national_integration_mobilization_speed_cap

		}
		tooltip = mobilization_speed_tooltip
	}

	# Division Organisation
	add_to_variable = {

	}

			subtract = AFG_national_integration_production_speed_buildings_factor
		}
		tooltip = production_speed_buildings_factor_tooltip
	}

	# Research Speed
	add_to_variable = {
		AFG_national_integration_research_speed_factor = {
			value = @AFG_national_integration_research_speed_factor_cap

# Nothing needs to be set to first run this effect
AFG_increase_national_integration = {
	if = {
		limit = { has_dynamic_modifier = { modifier = AFG_national_integration_modifier } }
		if = {
			limit = { check_expr = { value = AFG_reform_progress subtract = 1 equals = 0 } }
			remove_dynamic_modifier = { modifier = AFG_national_integration_modifier }
			clear_variable = AFG_reform_progress
			clear_variable = AFG_national_integration_mobilization_speed
			clear_variable = AFG_national_integration_army_org_factor
			clear_variable = AFG_national_integration_production_speed_buildings_factor
			clear_variable = AFG_national_integration_research_speed_factor
		}
		else = {
			custom_effect_tooltip = {
				localization_key = tooltip_modify_dynmod
				DYNMOD = AFG_national_integration_modifier
			}
			subtract_from_variable = { AFG_reform_progress = 1 }

			# Mobilisation Speed
			add_to_variable = {
				AFG_national_integration_mobilization_speed = {
					value = @AFG_national_integration_mobilization_speed_cap
					divide = @AFG_reform_progress_amount
				}
				tooltip = mobilization_speed_tooltip
			}

			# Division Organisation
			add_to_variable = {
				AFG_national_integration_army_org_factor = {
					value = @AFG_national_integration_army_org_factor_cap
					divide = @AFG_reform_progress_amount
				}
				tooltip = army_org_factor_tooltip
			}

			# Construction Speed
			add_to_variable = {
				AFG_national_integration_production_speed_buildings_factor = {
					value = @AFG_national_integration_production_speed_buildings_factor_cap
					divide = @AFG_reform_progress_amount
				}
				tooltip = production_speed_buildings_factor_tooltip
			}

			# Research Speed
			add_to_variable = {
				AFG_national_integration_research_speed_factor = {
					value = @AFG_national_integration_research_speed_factor_cap
					divide = @AFG_reform_progress_amount
				}
				tooltip = research_speed_factor_tooltip
			}
		}
	}
	else_if = {
		limit = { has_idea = AFG_industrial_revolution }
		swap_ideas = {
			remove_idea = AFG_industrial_revolution
			add_idea = AFG_industrial_revolution_2
		}
	}
	else_if = {
		limit = { NOT = { has_idea = AFG_industrial_revolution_2 } }
		add_ideas = AFG_industrial_revolution
	}
	else = {
		# Extra bonus in case there is still ability to do more
		add_one_random_civilian_factory = yes
	}
}

AFG_calculate_foreign_influence_share = {
	set_temp_variable = {
		AFG_foreign_influence_total = {
			value = AFG.AFG_foreign_influence_@GER
			add = AFG.AFG_foreign_influence_@RUS
			add = AFG.AFG_foreign_influence_@TUR
			add = AFG.AFG_foreign_influence_@JAP
		}
	}

	set_temp_variable = { AFG_foreign_influence_share_total = 1 }
	set_temp_variable = { AFG_highest_foreign_influence_amount = 0 }
	for_each_scope_loop = {
		array = AFG.AFG_foreign_influence_countries
		add_to_temp_variable = { AFG_loop_current = 1 }

		### Calculate each Respective Share ###
		if = {
			limit = { check_variable = { AFG.AFG_foreign_influence_countries^num = AFG_loop_current } }
			set_variable = { AFG.AFG_foreign_influence_share_@THIS = AFG_foreign_influence_share_total }
		}
		else = {
			set_variable = {
				AFG.AFG_foreign_influence_share_@THIS = {
					value = AFG.AFG_foreign_influence_@THIS
					divide = AFG_foreign_influence_total
					multiply = 100
					round = yes
					divide = 100
				}
			}
			subtract_from_temp_variable = { AFG_foreign_influence_share_total = AFG.AFG_foreign_influence_share_@THIS }
		}

		### Determine who has highest influence ###
		if = {
			limit = { check_variable = { AFG_highest_foreign_influence_amount < AFG.AFG_foreign_influence_@THIS } }
			set_temp_variable = { AFG_highest_foreign_influence_amount = AFG.AFG_foreign_influence_@THIS }
			set_variable = { AFG.AFG_highest_foreign_influence = THIS }
		}
	}
	AFG_calculate_foreign_investment_cost = yes
}

@AFG_foreign_influence_pp_ratio = 3 # 1 influence = 3 political power
# 25% influence share = x1 AFG_foreign_influence_share_multiplier
# 50% influence share = x2 AFG_foreign_influence_share_multiplier
# 75% influence share = x3 AFG_foreign_influence_share_multiplier
# 100% influence share = x4 AFG_foreign_influence_share_multiplier
AFG_calculate_foreign_investment_cost = {
	# Investment Cost Formula: (Influence * @AFG_foreign_influence_pp_ratio) * AFG_foreign_influence_share_multiplier
	# AFG_foreign_influence_share_multiplier Formula:
	# AFG_foreign_influence_share_multiplier = 1 | AFG_foreign_influence_share_@THIS <= 0.25
	# AFG_foreign_influence_share_multiplier = (AFG_foreign_influence_share_@THIS * 4) | AFG_foreign_influence_share_@THIS > 0.25
	for_each_scope_loop = {
		array = AFG_foreign_influence_countries
		set_variable = {
			AFG.AFG_foreign_investment_cost_@THIS = {
				value = AFG.AFG_foreign_influence_@THIS
				multiply = @AFG_foreign_influence_pp_ratio
				multiply = { #AFG_foreign_influence_share_multiplier
					value = AFG.AFG_foreign_influence_share_@THIS
					multiply = 4
					clamp = { min = 1 max = 4 }
				}
				round = yes
				clamp = { min = 10 max = 500 }
			}
		}
	}
}
