update_UFC_profits = {
	if = {
		limit = { check_variable = { cos_profit_change < 0 } }
		add_to_variable = { COS_remove_market_decisions = cos_profit_change }
	}
	if = {
		limit = { check_variable = { COS_remove_market_decisions < 1 } }
		remove_ideas = COS_UFC_Profit_idea
	}
	else = {
		add_to_variable = {
			cos_profit_factor = cos_profit_change
			tooltip = COS_UFC_profit_change_tt
		}
		custom_effect_tooltip = {
			localization_key = tooltip_modify_dynmod
			DYNMOD = COS_UFC_Profit_idea
		}
		add_to_variable = { COS_UFC_consumer_goods = { value = -0.015 multiply = cos_profit_change } tooltip = consumer_goods_factor_tooltip }
		add_to_variable = { COS_UFC_output = { value = 0.005 multiply = cos_profit_change } tooltip = COS_UFC_output_tooltip }
	}
	force_update_dynamic_modifier = yes
}

game_start_UFC_influence = {
	set_variable = { COS_UFC_influence_modifier = { value = COS_UFC_influence multiply = -0.005 } }
	set_variable = { COS_UFC_influence_construction_speed = { value = COS_UFC_influence multiply = 0.003 } }
	set_variable = { COS_UFC_influence_consumer_goods = { value = COS_UFC_influence multiply = -0.003 } }
}

update_UFC_influence = {
	if = {
		limit = { has_idea = COS_influence_of_the_UFC_idea }
		if = {
			limit = { check_variable = { cos_UFC_change > 0 } }
			custom_effect_tooltip = COS_UFC_plus_tt
		}
		else = {
			set_temp_variable = { cos_UFC_change_dummy = { value = cos_UFC_change multiply = -1 } }
			custom_effect_tooltip = COS_UFC_minus_tt
		}
		add_to_variable = {
			COS_UFC_influence_modifier = {
				value = cos_UFC_change multiply = -0.005
				clamp = { min = { value = -0.5 subtract = COS_UFC_influence_modifier } max = { value = COS_UFC_influence_modifier multiply = -1 } }
			}
		}
		add_to_variable = {
			COS_UFC_influence_construction_speed = {
				value = cos_UFC_change multiply = 0.003
				clamp = { min = { value = COS_UFC_influence_construction_speed multiply = -1 } max = { value = 0.3 subtract = COS_UFC_influence_construction_speed } }
			}
		}
		add_to_variable = {
			COS_UFC_influence_consumer_goods = {
				value = cos_UFC_change multiply = -0.003
				clamp = { min = { value = -0.3 subtract = COS_UFC_influence_consumer_goods } max = { value = COS_UFC_influence_consumer_goods multiply = -1 } }
			}
		}
		add_to_variable = {
			COS_UFC_influence = {
				value = cos_UFC_change
				clamp = { min = { value = COS_UFC_influence multiply = -1 } max = { value = 100 subtract = COS_UFC_influence } }
			}
		}
		force_update_dynamic_modifier = yes
	}
	else_if = {
		limit = { check_variable = { cos_UFC_change > 0 } }
		multiply_temp_variable = { cos_UFC_change = 15 }
		if = {
			limit = { has_idea = COS_investing }
			modify_timed_idea = {
				idea = COS_investing
				days = cos_UFC_change
			}
		}
		else = {
			add_timed_idea = {
				idea = COS_investing
				days = cos_UFC_change
			}
		}
	}
}
