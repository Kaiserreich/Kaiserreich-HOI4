AUS_crush_hungary = {
	allowed = {
		OR = {
			original_tag = AUS
			has_autonomy_state = crownland
		}
	}
	enable = {
		has_war_with = HUN
		has_civil_war = yes
		check_variable = { enemies^num = 1 } #Only at war with HUN

		#activate every 25 days, then pause for another 25
		check_expr = {
			value = global.num_days
			mod = 50
			less_than = 25
		}
	}
	abort_when_not_enabled = yes

	ai_strategy = {
		type = conquer
		id = HUN
		value = 100
	}
	ai_strategy = {
		type = strategic_air_importance
		id = 27 #Hungary
		value = 30000
	}
	ai_strategy = {
		type = front_control
		tag = HUN
		priority = 100
		ordertype = front
		execution_type = rush_weak
		execute_order = yes
	}
}

AUS_crownlands_if_safe_should_support_faction = {
	allowed = {
		OR = {
			original_tag = CZE
			original_tag = HUN
			original_tag = SLO
			original_tag = TRS
			original_tag = CRO
			original_tag = BOS
			original_tag = GAL
		}
	}
	enable = {
		has_war = yes
		OR = {
			GER = { surrender_progress > 0 }
			AUS = { surrender_progress > 0 }
		}
		NOT = { surrender_progress > 0 }
		NOT = { any_enemy_country = { is_neighbor_of = ROOT } }
		OR = {
			NOT = { any_owned_state = { is_coastal = yes } }
			is_subject = no						#the collapse of Austria should make them extra willing to do this
		}
	}
	abort_when_not_enabled = yes

	ai_strategy = {
		type = garrison
		value = -75
	}

	ai_strategy = {
		type = front_unit_request
		country_trigger = { NOT = { is_neighbor_of = FROM } }		#counteract 00_area_priority
		value = 25
	}
}
